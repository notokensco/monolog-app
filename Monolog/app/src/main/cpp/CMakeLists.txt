cmake_minimum_required(VERSION 3.22)
project(monolog_whisper LANGUAGES C CXX)

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${MONOLOG_WHISPER_SOURCE}" whisper-build EXCLUDE_FROM_ALL)

add_library(monolog_whisper SHARED whisper_bridge.cpp)
target_compile_features(monolog_whisper PRIVATE cxx_std_17)
target_link_options(monolog_whisper PRIVATE "-Wl,-z,max-page-size=16384")
target_link_libraries(monolog_whisper PRIVATE whisper)
